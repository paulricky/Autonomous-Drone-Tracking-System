cmake_minimum_required(VERSION 3.16)
project(ece3162_perception LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
add_library(stereo_synchronizer src/stereo_synchronizer.cpp)
target_include_directories(stereo_synchronizer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
install(TARGETS stereo_synchronizer EXPORT perceptionTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(perceptionTargets HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_stereo_synchronizer test/test_stereo_synchronizer.cpp)
  target_link_libraries(test_stereo_synchronizer stereo_synchronizer)
endif()
ament_package()
