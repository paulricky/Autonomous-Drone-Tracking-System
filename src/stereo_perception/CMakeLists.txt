cmake_minimum_required(VERSION 3.16)
project(stereo_perception LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d video videoio imgcodecs)
find_package(rclcpp REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(flight_interfaces REQUIRED)
add_library(stereo_synchronizer src/stereo_synchronizer.cpp src/stereo_pipeline.cpp src/synthetic_scene.cpp src/state_estimator.cpp)
target_include_directories(stereo_synchronizer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(stereo_synchronizer PUBLIC ${OpenCV_LIBS})
add_executable(perception_node src/perception_node.cpp)
target_link_libraries(perception_node stereo_synchronizer)
ament_target_dependencies(perception_node rclcpp cv_bridge sensor_msgs nav_msgs diagnostic_msgs flight_interfaces tf2_ros)
add_executable(uvc_stereo_node src/uvc_stereo_node.cpp)
add_executable(calibrate_stereo src/calibrate_stereo.cpp)
target_link_libraries(calibrate_stereo stereo_synchronizer)
target_link_libraries(uvc_stereo_node stereo_synchronizer)
ament_target_dependencies(uvc_stereo_node rclcpp cv_bridge sensor_msgs diagnostic_msgs)
install(TARGETS stereo_synchronizer perception_node uvc_stereo_node calibrate_stereo EXPORT perceptionTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(perceptionTargets HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_stereo_synchronizer test/test_stereo_synchronizer.cpp)
  target_link_libraries(test_stereo_synchronizer stereo_synchronizer)
  ament_add_gtest(test_stereo_pipeline test/test_stereo_pipeline.cpp)
  target_link_libraries(test_stereo_pipeline stereo_synchronizer)
endif()
ament_package()
