cmake_minimum_required(VERSION 3.16)
project(robotics_core LANGUAGES CXX)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(flight_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(${PROJECT_NAME}
  src/msp.cpp src/msp_serial.cpp src/command_gate.cpp src/trajectory_tracker.cpp src/voxel_map.cpp
  src/kinodynamic_astar.cpp src/bspline.cpp src/ego_optimizer.cpp src/dynamics.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
add_library(operator_session src/operator_session.cpp)
target_include_directories(operator_session PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(operator_session PUBLIC OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_executable(drone_supervisor_node src/drone_supervisor_node.cpp)
target_link_libraries(drone_supervisor_node operator_session)
ament_target_dependencies(drone_supervisor_node rclcpp std_msgs)
add_executable(operator_receiver_node src/operator_receiver_node.cpp)
target_link_libraries(operator_receiver_node operator_session)
ament_target_dependencies(operator_receiver_node rclcpp std_msgs geometry_msgs)
add_executable(hardware_monitor_node src/hardware_monitor_node.cpp)
add_executable(msp_telemetry_node src/msp_telemetry_node.cpp)
target_link_libraries(msp_telemetry_node ${PROJECT_NAME})
add_executable(physical_command_gate_node src/physical_command_gate_node.cpp)
target_link_libraries(physical_command_gate_node ${PROJECT_NAME})
target_link_libraries(hardware_monitor_node ${PROJECT_NAME})
add_executable(physical_navigation_node src/autonomy_sim_node.cpp)
target_compile_definitions(physical_navigation_node PRIVATE ROBOTICS_PHYSICAL)
target_link_libraries(physical_navigation_node ${PROJECT_NAME})
ament_target_dependencies(physical_navigation_node rclcpp rclcpp_action flight_interfaces std_msgs sensor_msgs geometry_msgs tf2_ros)
add_executable(px4_flight_adapter_node src/px4_flight_adapter_node.cpp)
target_link_libraries(px4_flight_adapter_node operator_session)
ament_target_dependencies(px4_flight_adapter_node rclcpp flight_interfaces std_msgs sensor_msgs mavros_msgs rcl_interfaces geometry_msgs nav_msgs)
add_executable(autonomy_sim_node src/autonomy_sim_node.cpp)
target_link_libraries(autonomy_sim_node ${PROJECT_NAME})
ament_target_dependencies(autonomy_sim_node rclcpp rclcpp_action flight_interfaces std_msgs sensor_msgs tf2_ros)
ament_target_dependencies(hardware_monitor_node rclcpp rclcpp_lifecycle diagnostic_msgs flight_interfaces)
ament_target_dependencies(msp_telemetry_node rclcpp diagnostic_msgs flight_interfaces)
ament_target_dependencies(physical_command_gate_node rclcpp diagnostic_msgs flight_interfaces)
install(TARGETS drone_supervisor_node operator_session operator_receiver_node physical_navigation_node px4_flight_adapter_node ${PROJECT_NAME} hardware_monitor_node msp_telemetry_node physical_command_gate_node autonomy_sim_node EXPORT ${PROJECT_NAME}Targets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(${PROJECT_NAME}Targets HAS_LIBRARY_TARGET)
ament_export_dependencies(OpenSSL nlohmann_json)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_native_operator_ros test/test_native_operator_ros.py
    ENV "NATIVE_OPERATOR_EXECUTABLE=$<TARGET_FILE:operator_receiver_node>" TIMEOUT 30)
  ament_add_pytest_test(test_px4_adapter_ros test/test_px4_adapter_ros.py
    ENV "NATIVE_ADAPTER_EXECUTABLE=$<TARGET_FILE:px4_flight_adapter_node>" TIMEOUT 30)
  ament_add_gtest(test_operator_session test/test_operator_session.cpp)
  target_link_libraries(test_operator_session operator_session)
  ament_add_gtest(test_msp test/test_msp.cpp)
  target_link_libraries(test_msp ${PROJECT_NAME})
  ament_add_gtest(test_msp_serial test/test_msp_serial.cpp)
  target_link_libraries(test_msp_serial ${PROJECT_NAME})
  ament_add_gtest(test_command_gate test/test_command_gate.cpp)
  target_link_libraries(test_command_gate ${PROJECT_NAME})
  ament_add_gtest(test_trajectory_tracker test/test_trajectory_tracker.cpp)
  target_link_libraries(test_trajectory_tracker ${PROJECT_NAME})
  ament_add_gtest(test_navigation test/test_navigation.cpp)
  target_link_libraries(test_navigation ${PROJECT_NAME})
endif()
ament_package()
